import { useEffect, useRef } from "react";
import { EditorState, StateEffect } from "@codemirror/state";
import { EditorView, lineNumbers } from "@codemirror/view";
import { loadLanguageExtension } from "../lib/codeLanguage";
import { codeEditorTheme, codeSyntaxHighlighting } from "../lib/codeTheme";

export default function ReadonlyCode({
  code,
  language,
  onLayout,
}: {
  code: string;
  language: string;
  onLayout?: () => void;
}) {
  const hostRef = useRef<HTMLDivElement>(null);
  const onLayoutRef = useRef(onLayout);
  onLayoutRef.current = onLayout;

  useEffect(() => {
    const host = hostRef.current;
    if (!host) {
      return;
    }

    const view = new EditorView({
      parent: host,
      state: EditorState.create({
        doc: code,
        extensions: [
          lineNumbers(),
          codeSyntaxHighlighting,
          codeEditorTheme,
          EditorView.lineWrapping,
          EditorState.readOnly.of(true),
          EditorView.editable.of(false),
        ],
      }),
    });

    const observer = new ResizeObserver(() => {
      onLayoutRef.current?.();
    });
    observer.observe(host);

    const notify = () => {
      requestAnimationFrame(() => onLayoutRef.current?.());
    };
    notify();

    void loadLanguageExtension(language)
      .then((extension) => {
        if (extension) {
          view.dispatch({ effects: StateEffect.appendConfig.of(extension) });
          notify();
        }
      })
      .catch(() => {
        // 不支持的语言保持纯文本。
      });

    return () => {
      observer.disconnect();
      view.destroy();
    };
  }, [code, language]);

  return <div ref={hostRef} className="readonly-code-mirror" />;
}

import { useEffect, useRef } from "react";

export default function TextRunEditor({
  startIndex,
  endIndex,
  initialText,
  initialCaret,
  saving,
  onSave,
  onDirty,
  onCancel,
}: {
  startIndex: number;
  endIndex: number;
  initialText: string;
  initialCaret?: number;
  saving: boolean;
  onSave: (start: number, end: number, text: string) => void | Promise<void>;
  onDirty: () => void;
  onCancel: () => void;
}) {
  const ref = useRef<HTMLTextAreaElement>(null);
  const committed = useRef(false);
  const dirty = useRef(false);
  const composing = useRef(false);

  function resize() {
    const node = ref.current;
    if (!node) {
      return;
    }
    node.style.height = "auto";
    node.style.height = `${Math.max(30, node.scrollHeight)}px`;
  }

  useEffect(() => {
    const node = ref.current;
    if (!node) {
      return;
    }
    resize();
    node.focus();
    const caret = Math.min(
      initialCaret ?? node.value.length,
      node.value.length,
    );
    node.setSelectionRange(caret, caret);
  }, []);

  function commit() {
    const node = ref.current;
    if (!node || committed.current || composing.current) {
      return;
    }
    committed.current = true;
    void onSave(startIndex, endIndex, node.value);
  }

  return (
    <div className="text-run-editor">
      <textarea
        ref={ref}
        className="block-textarea text-run-textarea"
        defaultValue={initialText}
        rows={1}
        spellCheck={false}
        disabled={saving}
        onCompositionStart={() => {
          composing.current = true;
        }}
        onCompositionEnd={() => {
          composing.current = false;
          resize();
        }}
        onInput={() => {
          resize();
          if (!dirty.current) {
            dirty.current = true;
            onDirty();
          }
        }}
        onBlur={commit}
        onKeyDown={(event) => {
          const node = event.currentTarget;
          if (event.key === "Escape") {
            event.preventDefault();
            onCancel();
            return;
          }
          if (event.key === "Enter" && (event.ctrlKey || event.metaKey)) {
            event.preventDefault();
            node.blur();
            return;
          }
          if (
            event.key === "Enter" &&
            !event.altKey &&
            !composing.current
          ) {
            event.preventDefault();
            const start = node.selectionStart;
            const end = node.selectionEnd;
            const lineStart = node.value.lastIndexOf("\n", start - 1) + 1;
            const currentLine = node.value.slice(lineStart, start);
            // 已经位于空段落时只前进一行；否则 Enter 新建段落，Shift+Enter 软换行。
            const insertion =
              event.shiftKey || currentLine.trim() === ""
                ? "\n"
                : "\n\n";
            node.setRangeText(insertion, start, end, "end");
            resize();
            if (!dirty.current) {
              dirty.current = true;
              onDirty();
            }
          }
        }}
      />
    </div>
  );
}

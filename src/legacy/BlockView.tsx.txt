import type { ReactNode } from "react";
import type { Block } from "../types";
import ReadonlyCode from "./ReadonlyCode";
import { parseCodeBlock } from "../lib/codeBlock";

const INLINE = /(\*\*[^*]+\*\*|__[^_]+__|\*[^*]+\*|`[^`]+`|\[[^\]]+\]\([^)]+\))/g;

function inline(text: string): ReactNode[] {
  const parts = text.split(INLINE).filter(Boolean);
  return parts.map((part, index) => {
    if (part.startsWith("**") && part.endsWith("**")) {
      return <strong key={index}>{part.slice(2, -2)}</strong>;
    }
    if (part.startsWith("__") && part.endsWith("__")) {
      return <strong key={index}>{part.slice(2, -2)}</strong>;
    }
    if (part.startsWith("*") && part.endsWith("*")) {
      return <em key={index}>{part.slice(1, -1)}</em>;
    }
    if (part.startsWith("`") && part.endsWith("`")) {
      return <code key={index}>{part.slice(1, -1)}</code>;
    }
    const link = part.match(/^\[([^\]]+)\]\(([^)]+)\)$/);
    if (link) {
      return (
        <a key={index} href={link[2]} target="_blank" rel="noreferrer">
          {link[1]}
        </a>
      );
    }
    return <span key={index}>{part}</span>;
  });
}

function headingLevel(text: string): number {
  return Math.min(6, text.match(/^#+/)?.[0].length ?? 1);
}

function headingText(text: string): string {
  return text.replace(/^#+\s*/, "").replace(/\s+#+\s*$/, "");
}

function listItems(text: string): string[] {
  return text
    .split("\n")
    .map((line) => line.replace(/^\s*(?:[-*+]|\d+[.)])\s+/, ""))
    .filter((line) => line.length > 0);
}

function tableRows(text: string): string[][] {
  const rows = text
    .split("\n")
    .map((line) => line.trim())
    .filter((line) => line.startsWith("|"))
    .map((line) =>
      line
        .replace(/^\|/, "")
        .replace(/\|$/, "")
        .split("|")
        .map((cell) => cell.trim()),
    );
  return rows.filter((row) => !row.every((cell) => /^:?-{3,}:?$/.test(cell)));
}

export default function BlockView({
  block,
  onCodeLayout,
}: {
  block: Block;
  onCodeLayout?: () => void;
}) {
  switch (block.kind) {
    case "heading": {
      const level = headingLevel(block.text);
      const Tag = `h${level}` as "h1" | "h2" | "h3" | "h4" | "h5" | "h6";
      return <Tag className="md-heading">{inline(headingText(block.text))}</Tag>;
    }
    case "list":
      return (
        <ul className="md-list">
          {listItems(block.text).map((item, index) => (
            <li key={index}>{inline(item)}</li>
          ))}
        </ul>
      );
    case "quote":
      return (
        <blockquote className="md-quote">
          {inline(block.text.replace(/^>\s?/gm, ""))}
        </blockquote>
      );
    case "code": {
      const parts = parseCodeBlock(block.text);
      return (
        <div className="md-code">
          <ReadonlyCode
            code={parts.code}
            language={parts.language}
            onLayout={onCodeLayout}
          />
        </div>
      );
    }
    case "table": {
      const rows = tableRows(block.text);
      const [header, ...body] = rows;
      return (
        <div className="md-table-wrap">
          <table className="md-table">
            {header && (
              <thead>
                <tr>
                  {header.map((cell, index) => (
                    <th key={index}>{inline(cell)}</th>
                  ))}
                </tr>
              </thead>
            )}
            <tbody>
              {body.map((row, rowIndex) => (
                <tr key={rowIndex}>
                  {row.map((cell, cellIndex) => (
                    <td key={cellIndex}>{inline(cell)}</td>
                  ))}
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      );
    }
    case "rule":
      return <hr className="md-rule" />;
    default:
      return <p className="md-paragraph">{inline(block.text)}</p>;
  }
}

import { useEffect, useMemo, useRef, useState } from "react";
import { useVirtualizer } from "@tanstack/react-virtual";
import type { Block, BlockDocument } from "../types";
import BlockView from "./BlockView";
import BlockEditor from "./BlockEditor";
import CodeBlockEditor from "./CodeBlockEditor";
import TextRunEditor from "./TextRunEditor";
import { parseCodeBlock } from "../lib/codeBlock";

type RunItem = {
  block: Block;
  index: number;
};

type Row =
  | {
      id: string;
      kind: "text";
      items: RunItem[];
      startIndex: number;
      endIndex: number;
    }
  | {
      id: string;
      kind: "special";
      block: Block;
      index: number;
    };

function buildRows(blocks: Block[]): Row[] {
  const rows: Row[] = [];
  let index = 0;
  while (index < blocks.length) {
    if (blocks[index].kind !== "paragraph") {
      rows.push({
        id: `block:${blocks[index].id}`,
        kind: "special",
        block: blocks[index],
        index,
      });
      index += 1;
      continue;
    }

    const items: RunItem[] = [];
    while (index < blocks.length && blocks[index].kind === "paragraph") {
      items.push({ block: blocks[index], index });
      index += 1;
    }
    const real = items.filter((item) => !item.block.id.startsWith("new_"));
    rows.push({
      id: `run:${items[0].block.id}`,
      kind: "text",
      items,
      startIndex: real[0]?.index ?? -1,
      endIndex: real[real.length - 1]?.index ?? -1,
    });
  }
  return rows;
}

function runText(items: RunItem[]): string {
  return items.map((item) => item.block.text).join("\n\n");
}

function caretAtIndex(items: RunItem[], clickedIndex: number): number {
  let offset = 0;
  for (const item of items) {
    offset += item.block.text.length;
    if (item.index === clickedIndex) {
      return offset;
    }
    offset += 2;
  }
  return offset;
}

export default function VirtualDocument({
  document,
  editingId,
  saving,
  selected,
  onStartEdit,
  onSelectRange,
  onDirty,
  onSave,
  onSplitBlock,
  onMergePrevious,
  onSaveTextRun,
  onNewBlock,
  onCancel,
}: {
  document: BlockDocument;
  editingId: string | null;
  saving: boolean;
  selected: number[];
  onStartEdit: (id: string, index: number) => void;
  onSelectRange: (index: number) => void;
  onDirty: () => void;
  onSave: (blockId: string, index: number, text: string) => void | Promise<void>;
  onSplitBlock: (
    blockId: string,
    index: number,
    before: string,
    after: string,
  ) => void | Promise<void>;
  onMergePrevious: (index: number, text: string) => void | Promise<void>;
  onSaveTextRun: (
    startIndex: number,
    endIndex: number,
    text: string,
  ) => void | Promise<void>;
  onNewBlock: () => void;
  onCancel: () => void;
}) {
  const parentRef = useRef<HTMLDivElement>(null);
  const copyTimer = useRef<number | null>(null);
  const [copiedId, setCopiedId] = useState<string | null>(null);
  const [caretOffset, setCaretOffset] = useState<number | undefined>(undefined);
  const rows = useMemo(() => buildRows(document.blocks), [document.blocks]);
  const virtualizer = useVirtualizer({
    count: rows.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 96,
    overscan: 8,
  });

  useEffect(() => {
    const parent = parentRef.current;
    if (!parent) {
      return;
    }
    parent.scrollTop = 0;
    virtualizer.scrollToOffset(0);
    const frame = requestAnimationFrame(() => {
      parent.scrollTop = 0;
      virtualizer.scrollToOffset(0);
    });
    return () => cancelAnimationFrame(frame);
    // 切换文档时回到顶部。
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [document.nodeId]);

  useEffect(() => {
    if (!editingId) {
      return;
    }
    const rowIndex = rows.findIndex((row) =>
      row.kind === "text"
        ? row.items.some((item) => item.block.id === editingId)
        : row.block.id === editingId,
    );
    if (rowIndex >= 0) {
      const frame = requestAnimationFrame(() => {
        virtualizer.scrollToIndex(rowIndex, { align: "auto" });
      });
      return () => cancelAnimationFrame(frame);
    }
    // 只在切换编辑块时调整视口。
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [editingId]);

  useEffect(() => {
    return () => {
      if (copyTimer.current !== null) {
        window.clearTimeout(copyTimer.current);
      }
    };
  }, []);

  async function copyCode(blockId: string, text: string) {
    const code = parseCodeBlock(text).code;
    try {
      await navigator.clipboard.writeText(code);
    } catch {
      const textarea = window.document.createElement("textarea");
      textarea.value = code;
      textarea.style.position = "fixed";
      textarea.style.opacity = "0";
      window.document.body.appendChild(textarea);
      textarea.select();
      window.document.execCommand("copy");
      textarea.remove();
    }

    setCopiedId(blockId);
    if (copyTimer.current !== null) {
      window.clearTimeout(copyTimer.current);
    }
    copyTimer.current = window.setTimeout(() => {
      setCopiedId((current) => (current === blockId ? null : current));
    }, 1400);
  }

  function startTextEdit(row: Extract<Row, { kind: "text" }>, clicked: RunItem) {
    setCaretOffset(caretAtIndex(row.items, clicked.index));
    const firstReal = row.items.find((item) => !item.block.id.startsWith("new_"));
    const target = firstReal ?? row.items[0];
    onStartEdit(target.block.id, target.index);
  }

  return (
    <div ref={parentRef} className="virtual-scroll">
      <div
        className="virtual-inner"
        style={{ height: `${virtualizer.getTotalSize()}px` }}
      >
        {virtualizer.getVirtualItems().map((virtualItem) => {
          const row = rows[virtualItem.index];
          if (!row) {
            return null;
          }
          const editing =
            row.kind === "text"
              ? row.items.some((item) => item.block.id === editingId)
              : row.block.id === editingId;
          const isSelected =
            row.kind === "text"
              ? row.items.some((item) => selected.includes(item.index))
              : selected.includes(row.index);
          const rowIndex = virtualItem.index;

          if (row.kind === "text") {
            const hasReal = row.startIndex >= 0;
            return (
              <div
                key={row.id}
                data-index={rowIndex}
                ref={(node) => {
                  if (node) {
                    virtualizer.measureElement(node);
                  }
                }}
                className={`virtual-block text-run-block ${
                  editing ? "editing" : ""
                } ${isSelected ? "selected" : ""}`}
                style={{ transform: `translateY(${virtualItem.start}px)` }}
                onClick={(event) => {
                  if (event.shiftKey) {
                    onSelectRange(row.items[0].index);
                  } else if (!editing) {
                    startTextEdit(row, row.items[row.items.length - 1]);
                  }
                }}
              >
                {editing ? (
                  hasReal ? (
                    <TextRunEditor
                      startIndex={row.startIndex}
                      endIndex={row.endIndex}
                      initialText={runText(row.items)}
                      initialCaret={caretOffset}
                      saving={saving}
                      onSave={onSaveTextRun}
                      onDirty={onDirty}
                      onCancel={onCancel}
                    />
                  ) : (
                    <BlockEditor
                      block={row.items[0].block}
                      saving={saving}
                      onDirty={onDirty}
                      onSave={(text) =>
                        onSave(row.items[0].block.id, row.items[0].index, text)
                      }
                      onEnter={(before, after) =>
                        onSplitBlock(
                          row.items[0].block.id,
                          row.items[0].index,
                          before,
                          after,
                        )
                      }
                      onNavigate={() => {}}
                      onMergePrevious={() => {}}
                      onCancel={onCancel}
                    />
                  )
                ) : (
                  <div className="text-run">
                    {row.items.map((item) => (
                      <BlockView key={item.block.id} block={item.block} />
                    ))}
                  </div>
                )}
              </div>
            );
          }

          const block = row.block;
          return (
            <div
              key={row.id}
              data-index={rowIndex}
              ref={(node) => {
                if (node) {
                  virtualizer.measureElement(node);
                }
              }}
              className={`virtual-block ${editing ? "editing" : ""} ${
                isSelected ? "selected" : ""
              }`}
              style={{ transform: `translateY(${virtualItem.start}px)` }}
              onMouseDown={(event) => {
                if (block.kind === "code" && !editing && !event.shiftKey) {
                  onStartEdit(block.id, row.index);
                }
              }}
              onClick={(event) => {
                if (event.shiftKey) {
                  onSelectRange(row.index);
                } else if (!editing) {
                  onStartEdit(block.id, row.index);
                }
              }}
            >
              {editing ? (
                block.kind === "code" ? (
                  <CodeBlockEditor
                    block={block}
                    onDirty={onDirty}
                    onSave={(text) => onSave(block.id, row.index, text)}
                    onCancel={onCancel}
                  />
                ) : (
                  <BlockEditor
                    block={block}
                    saving={saving}
                    onDirty={onDirty}
                    onSave={(text) => onSave(block.id, row.index, text)}
                    onEnter={(before, after) =>
                      onSplitBlock(block.id, row.index, before, after)
                    }
                    onNavigate={(direction, text) => {
                      const targetIndex =
                        direction === "prev" ? row.index - 1 : row.index + 1;
                      const target = document.blocks[targetIndex];
                      if (!target) {
                        if (direction === "next") {
                          void Promise.resolve(
                            onSave(block.id, row.index, text),
                          ).then(onNewBlock);
                        }
                        return;
                      }
                      void Promise.resolve(
                        onSave(block.id, row.index, text),
                      ).then(() => onStartEdit(target.id, targetIndex));
                    }}
                    onMergePrevious={(text) =>
                      onMergePrevious(row.index, text)
                    }
                    onCancel={onCancel}
                  />
                )
              ) : block.kind === "code" ? (
                <div className="code-readonly">
                  <button
                    type="button"
                    className="code-copy-button"
                    title="复制围栏内的代码"
                    onMouseDown={(event) => event.stopPropagation()}
                    onClick={(event) => {
                      event.stopPropagation();
                      void copyCode(block.id, block.text);
                    }}
                  >
                    {copiedId === block.id ? "已复制" : "复制代码"}
                  </button>
                  <BlockView
                    block={block}
                    onCodeLayout={() => virtualizer.measure()}
                  />
                </div>
              ) : (
                <BlockView
                  block={block}
                  onCodeLayout={() => virtualizer.measure()}
                />
              )}
            </div>
          );
        })}
      </div>
      {!editingId && (
        <button type="button" className="new-block-button" onClick={onNewBlock}>
          + 新建块
        </button>
      )}
    </div>
  );
}
